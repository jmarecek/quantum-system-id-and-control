% ---------------------------------------------------------------------
% Cross-chapter references in single-lecture builds.
% Chapter files are compiled standalone with the subfiles package, so
% \ref{C.Closed} etc. would print "??" and trigger warnings.  When the
% .aux file does not define a chapter label, fall back to the chapter
% number fixed by the syllabus.  In the full builds the real labels win.
% ---------------------------------------------------------------------
\makeatletter
\newcommand*{\qocFallbackLabel}[2]{%
  \@ifundefined{r@#1}{\global\@namedef{r@#1}{{#2}{}{}{}{}}}{}}
\AtBeginDocument{%
  \qocFallbackLabel{C.Motivation}{1}%
  \qocFallbackLabel{C.Closed}{2}%
  \qocFallbackLabel{C.Open}{3}%
  \qocFallbackLabel{C.StateSpace}{4}%
  \qocFallbackLabel{C.Frequency}{5}%
  \qocFallbackLabel{C.SampleComplexity}{6}%
  \qocFallbackLabel{C.GeometryStates}{7}%
  \qocFallbackLabel{C.GeometryChannels}{8}%
  \qocFallbackLabel{C.GeometryControl}{9}%
  \qocFallbackLabel{C.Complexity}{10}%
  \qocFallbackLabel{C.AlgTomography}{11}%
  \qocFallbackLabel{C.AlgControl}{12}%
  \qocFallbackLabel{C.Revisited}{13}%
}
\makeatother

% ---------------------------------------------------------------------
% Cross-chapter references to theorems, equations and figures in
% single-lecture builds: import the labels of the last full build
% (labels_handouts.tex / labels_slides.tex, i.e. the \newlabel lines of
% handoutsbeamer.aux / overheadsbeamer.aux, extracted by make handouts,
% make slides or ./build.sh labels) for every label that the standalone
% .aux does not define.  Anchors are blanked, so hyperref prints the
% number without a dangling link.  Nothing happens in the full builds,
% and nothing breaks if the label files do not exist yet.
% ---------------------------------------------------------------------
\makeatletter
% A label value is normally {{ref}{page}{name}{anchor}{}}; amsbook also writes
% bare values such as \newlabel{tocindent0}{61.69456pt}, which are skipped.
\def\qoc@providelabel#1#2{\qoc@providelabel@i{#1}#2{}{}\qoc@end}
\def\qoc@providelabel@i#1{\@ifnextchar\bgroup{\qoc@providelabel@ii{#1}}{\qoc@gobbletoend}}
\def\qoc@gobbletoend#1\qoc@end{}
\def\qoc@providelabel@ii#1#2#3#4\qoc@end{%
  \@ifundefined{r@#1}{\global\@namedef{r@#1}{{#2}{#3}{}{}{}}}{}}
\newcommand*{\qocImportLabels}[1]{%
  \begingroup
    \makeatletter
    \let\newlabel\qoc@providelabel
    \InputIfFileExists{#1}{}{}%
  \endgroup}
\AtBeginDocument{%
  \ifnum\pdfstrcmp{\jobname}{handoutsbeamer}=0 \else
  \ifnum\pdfstrcmp{\jobname}{overheadsbeamer}=0 \else
  \ifnum\pdfstrcmp{\jobname}{book}=0 \else
    \only<article>{\qocImportLabels{labels_handouts.tex}}%
    \only<presentation>{\qocImportLabels{labels_slides.tex}\qocImportLabels{labels_handouts.tex}}% labels inside \ona{} exist only in the handouts
  \fi\fi\fi}
\makeatother
