\begin{frame}\ft{Outline}
\ona{\Chap~\ref{C.GeometryStates} described the set of states; this \chap{} describes the set of the maps between them, the channels, which an open quantum system implements between two instants. We met channels in \Chap~\ref{C.Open}, where we took the Kraus representation on trust and drew what three kinds of noise do to the Bloch ball. Here we prove the representation, by way of the matrix of Choi, and find that the channels, like the states, form a spectrahedron, now one level up; for a qubit the unital channels make a tetrahedron which, through the Choi matrix, is the tetrahedron of Bell-diagonal states of \Chap~\ref{C.GeometryStates}. We then ask how far apart two channels are. The answer depends on whether one asks about the average case, as the fidelities used in control do, or about the worst case, as the diamond norm does, and the two can differ by orders of magnitude. Channels can also be composed, and the question of which families of channels arise from a master equation with non-negative rates, and of what Markovian should mean for a quantum process, is a question about dividing a channel into channels. We close with the two algebraic structures that sit inside the set of channels: the Lie group of the unitary channels, on which \Chaps~\ref{C.GeometryControl} and~\ref{C.Complexity} build, and the Lie semigroup of the Markovian ones, which is not convex.}

\section{Channels and their representations}\label{sec:ch08:rep}

\begin{frame}\ft{Channels recalled}
\ona{A channel, by Definition~\ref{def:ch03:channel}, is a linear map $\Phi$ on the operators of $\Hil$ that preserves the trace and is completely positive: $\Phi\otimes\mathrm{id}_d$ maps positive semidefinite operators to positive semidefinite operators for every $d$. The reduced dynamics of a system that starts uncorrelated with its environment, $\rho_S(t)=\Tr_B\big[U(t)(\rho_S(0)\otimes\rho_B)U(t)^\dagger\big]$, is a channel for every $t$, and we write it $\Phi_t$, or $\mathcal E_{t,s}$ for the map from the time $s$ to the time $t$. If some of the population leaves the space we model, as when a transmon leaks into its third level (\Chap~\ref{C.Open}), the map is only trace non-increasing, $\Tr\Phi(\rho)\le\Tr\rho$; such maps are sometimes called quantum operations, and everything below holds for them with the obvious changes. Complete positivity cannot be weakened to positivity. The transposition $\rho\mapsto\rho^\top$ maps states to states, but applied to one half of the Bell state $\ket{\Phi^+}$ it produces the operator $\tfrac12\mathrm{SWAP}$, which has the eigenvalue $-\tfrac12$; a physical process acting on one of two entangled systems must leave the pair in a state.}
\onp{\begin{itemize}\setlength\itemsep{0.5em}
    \item Channel: linear, trace preserving, completely positive (Definition~\ref{def:ch03:channel}); $\Phi_t$ or $\mathcal E_{t,s}$.
    \item Leakage (\Chap~\ref{C.Open}): trace non-increasing maps, ``quantum operations''.
    \item Positivity is not enough: transposition on half of $\ket{\Phi^+}$ gives $\tfrac12\mathrm{SWAP}$, eigenvalue $-\tfrac12$.
\end{itemize}}
\end{frame}

\begin{frame}\ft{The Choi matrix}
\ona{A linear map on $N\times N$ matrices is fixed by its values on the matrix units $\ket i\bra j$, and these can be collected in a single matrix. With $\ket\Omega=\sum_i\ket{ii}$, the unnormalised maximally entangled vector, the \emph{Choi matrix} of $\Phi$ is}
\onp{Choi matrix, $\ket\Omega=\sum_i\ket{ii}$:}
\begin{equation}\label{eq:ch08:choi}
    \Choi\Phi\coloneqq\sum_{i,j=1}^N\ket i\bra j\otimes\Phi(\ket i\bra j)=(\mathrm{id}\otimes\Phi)\big(\ket\Omega\bra\Omega\big),
\end{equation}
\ona{the result of sending one half of a maximally entangled pair through the channel (Figure~\ref{fig:ch08:choi}(a)). The correspondence $\Phi\mapsto\Choi\Phi$ is a linear bijection between maps and $N^2\times N^2$ matrices; it is due to \citet{jamiolkowski1972linear} and \citet{CHOI1975285}, and the action of the map is recovered as $\Phi(\rho)=\Tr_1\big[(\rho^\top\otimes\Id)\Choi\Phi\big]$.}
\begin{thm}[Choi]\label{thm:ch08:choi}
$\Phi$ is completely positive if and only if $\Choi\Phi\succeq0$, and trace preserving if and only if $\Tr_2\Choi\Phi=\Id$. A completely positive $\Phi$ has a Kraus representation $\Phi(\rho)=\sum_{k=1}^\ell K_k\rho K_k^\dagger$ with $\ell=\rank\Choi\Phi\le N^2$, and $\sum_kK_k^\dagger K_k=\Id$ if $\Phi$ is trace preserving.
\end{thm}
\ona{\begin{pf}
If $\Phi$ is completely positive, $\Choi\Phi=(\mathrm{id}\otimes\Phi)(\ket\Omega\bra\Omega)\succeq0$ by definition. Conversely, let $\Choi\Phi=\sum_k\ket{v_k}\bra{v_k}$ be a decomposition into $\ell=\rank\Choi\Phi$ vectors, for instance the eigenvectors scaled by the square roots of the eigenvalues, and define $K_k$ by $\braket{m|K_k|i}=\big(\bra i\otimes\bra m\big)\ket{v_k}$, so that $\ket{v_k}=(\Id\otimes K_k)\ket\Omega$. Then $\sum_k(\Id\otimes K_k)\ket\Omega\bra\Omega(\Id\otimes K_k^\dagger)=\Choi\Phi$, and since a map is fixed by its Choi matrix, $\Phi(\rho)=\sum_kK_k\rho K_k^\dagger$. A map of this form is completely positive, since $(\mathrm{id}_d\otimes\Phi)(X)=\sum_k(\Id\otimes K_k)X(\Id\otimes K_k)^\dagger\succeq0$ for $X\succeq0$. Finally, $\Tr_2\Choi\Phi=\sum_{ij}\ket i\bra j\Tr\Phi(\ket i\bra j)$, which is $\Id$ if and only if $\Tr\Phi(\ket i\bra j)=\delta_{ij}$, that is, if and only if $\Phi$ preserves the trace; in terms of the Kraus operators this is $\sum_kK_k^\dagger K_k=\Id$.
\end{pf}}
\onp{\begin{itemize}
    \item $\Phi(\rho)=\Tr_1[(\rho^\top\otimes\Id)\Choi\Phi]$ \citep{jamiolkowski1972linear,CHOI1975285}.
    \item Proof of the Kraus form: eigenvectors of $\Choi\Phi$, reshaped.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Stinespring dilations and the freedom in the Kraus operators}
\ona{Collect the Kraus operators into one operator $V=\sum_kK_k\otimes\ket k_E$ from $\Hil$ into $\Hil\otimes\Hil_E$, where $\Hil_E$ is an auxiliary space with the orthonormal basis $\ket k_E$. Trace preservation, $\sum_kK_k^\dagger K_k=\Id$, says that $V^\dagger V=\Id$, so that $V$ is an isometry, and}
\onp{Stinespring: $V=\sum_kK_k\otimes\ket k_E$, $V^\dagger V=\Id$ \citep{stinespring1955positive}:}
\begin{equation}\label{eq:ch08:stinespring}
    \Phi(\rho)=\Tr_E\big[V\rho V^\dagger\big].
\end{equation}
\ona{This is the \emph{Stinespring dilation} \citep{stinespring1955positive}. Every channel is thus an isometric embedding of the system into a larger one, which can always be completed to a unitary evolution of the system together with an environment of dimension at most $N^2$, followed by discarding the environment: the system--bath picture of \Chap~\ref{C.Open} is not an assumption but a theorem. The Kraus operators are not unique. If $u$ is a unitary on $\Hil_E$, the operators $L_k=\sum_lu_{kl}K_l$ define the same channel, since $(\Id\otimes u)V$ gives the same partial trace over $E$; conversely, any two Kraus representations of one channel are related in this way, after padding the shorter with zeros. In pictures the freedom is a unitary and its inverse inserted on the environment leg (Figure~\ref{fig:ch08:choi}(b)), as the gauge freedom of a matrix product state was an invertible matrix and its inverse inserted on a bond (Figure~\ref{fig:ch07:tn:cut}). What is unique is the span of the Kraus operators, and their least number, the \emph{Kraus rank} $\rank\Choi\Phi$: one for a unitary channel, and at most $N^2$ (Figure~\ref{fig:ch08:choi}(c)).}
\onp{\begin{itemize}
    \item Every channel $=$ unitary on system $+$ environment ($\dim\le N^2$), then discard: \Chap~\ref{C.Open} as a theorem.
    \item Freedom: $L_k=\sum_lu_{kl}K_l$, a unitary on the environment leg, like the gauge of an MPS (\Chap~\ref{C.GeometryStates}).
    \item Kraus rank $=\rank\Choi\Phi\in\{1,\dots,N^2\}$; $1$ for a unitary channel.
\end{itemize}}
\end{frame}

\section{The convex body of channels}\label{sec:ch08:body}

\begin{frame}\ft{Channels form a spectrahedron}\mop{\small}
\ona{By Theorem~\ref{thm:ch08:choi}, the set $\Chan[\Hil]$ of channels on $\Hil=\C^N$ is affinely isomorphic to}
\begin{equation}\label{eq:ch08:slice}
    \{J\in\Herm[\C^N\otimes\C^N]:\ J\succeq0,\ \Tr_2J=\Id\},
\end{equation}
\ona{the intersection of the cone of positive semidefinite $N^2\times N^2$ matrices with an affine subspace of codimension $N^2$: a spectrahedron of real dimension $N^4-N^2$, which is $12$ for a qubit, the nine entries of the matrix $M$ and the three of the vector $\vec c$ in the affine map $\Bloch\mapsto M\Bloch+\vec c$ of the Bloch vector (\Chap~\ref{C.Open}). The normalised Choi matrices $\Choi\Phi/N$ are density operators of $\Hil\otimes\Hil$ whose reduced state on the first factor is maximally mixed, so that the channels are a slice of the states of a system twice as large. Optimising a linear function over channels, such as the fidelity with a target over all channels compatible with some data, is therefore a semidefinite programme, and the conic optimisation of \Chap~\ref{C.Motivation} returns in process tomography (\Chap~\ref{C.AlgTomography}). The \emph{process matrix} $\chi$ of that \chap, defined by $\Phi(\rho)=\sum_{mn}\chi_{mn}P_m\rho P_n^\dagger$ for a basis $\{P_m\}$ of Pauli strings, is the Choi matrix written in another basis.}
\onp{\begin{itemize}
    \item $\Chan[\C^N]\cong\{J\succeq0,\ \Tr_2J=\Id\}$: a spectrahedron of dimension $N^4-N^2$ ($12$ for a qubit).
    \item $\Choi\Phi/N$: states of $\Hil\otimes\Hil$ with maximally mixed marginal. Linear objectives: SDPs (\Chap~\ref{C.Motivation}).
    \item Process matrix $\chi$ (\Chap~\ref{C.AlgTomography}) $=$ Choi matrix in the Pauli basis.
\end{itemize}}
\begin{thm}[Choi; extreme points]\label{thm:ch08:extreme}
A channel with linearly independent Kraus operators $K_1,\dots,K_\ell$ is an extreme point of $\Chan[\Hil]$ if and only if the $\ell^2$ operators $K_j^\dagger K_k$ are linearly independent. Hence an extreme channel has Kraus rank at most $N$, and every unitary channel is extreme.
\end{thm}
\ona{Unlike the extreme points of $\Dens$, which are all unitarily equivalent pure states, those of $\Chan$ are of several kinds: the unitary channels, but also, for instance, amplitude damping, whose two Kraus operators $\operatorname{diag}(1,\sqrt{1-\gamma})$ and $\sqrt\gamma\ket0\bra1$ give four linearly independent products $K_j^\dagger K_k$ for $0<\gamma<1$.}
\onp{Extreme channels: unitaries, but also amplitude damping; Kraus rank $\le N$.}
\end{frame}

\begin{frame}\ft{Unital qubit channels: a tetrahedron}
\ona{For a qubit the body is twelve-dimensional, but its most important part can be drawn. A channel is \emph{unital} if $\Phi(\Id)=\Id$; for a qubit this means $\vec c=0$, so that the channel acts linearly on Bloch vectors, $\Bloch\mapsto M\Bloch$. Writing $M=R_1\operatorname{diag}(\lambda_1,\lambda_2,\lambda_3)R_2$ with rotations $R_1$, $R_2$, which are unitary channels, reduces every unital qubit channel to a diagonal one, and the diagonal ones are the \emph{Pauli channels} $\Phi(\rho)=\sum_{i=0}^3p_i\sigma_i\rho\sigma_i$, with $\lambda_1=p_0+p_1-p_2-p_3$, $\lambda_2=p_0-p_1+p_2-p_3$ and $\lambda_3=p_0-p_1-p_2+p_3$ \citep{king2001minimal}. In the coordinates $\vec\lambda$ the picture is complete (Figure~\ref{fig:ch08:pauli}(a)). The map is positive if and only if the ellipsoid with the semi-axes $\abs{\lambda_i}$ lies in the ball, that is, if $\vec\lambda$ lies in the cube $[-1,1]^3$. It is completely positive if and only if $\abs{\lambda_1\pm\lambda_2}\le\abs{1\pm\lambda_3}$ \citep{fujiwara1999one,ruskai2002analysis}, that is, if $\vec\lambda$ lies in the tetrahedron whose vertices are the four unitary channels $\Id$, $\sigma_x$, $\sigma_y$, $\sigma_z$, at the corners of the cube with $\lambda_1\lambda_2\lambda_3=1$. The other four corners, with $\lambda_1\lambda_2\lambda_3=-1$, are positive but not completely positive; one of them is the transposition $(1,-1,1)$. The Choi matrix of a Pauli channel is a Bell-diagonal state, with $\vec t=(\lambda_1,-\lambda_2,\lambda_3)$ in the coordinates \eqref{eq:ch07:belldiag}, so that this tetrahedron is the tetrahedron of states of Figure~\ref{fig:ch07:bell}, reflected. The separable octahedron of that figure becomes the octahedron $\abs{\lambda_1}+\abs{\lambda_2}+\abs{\lambda_3}\le1$ of the \emph{entanglement-breaking} channels, those whose output is separable from any ancilla, or equivalently those that can be implemented by measuring the input and preparing a state according to the outcome \citep{horodecki2003entanglement,ruskai2003qubit}. Every unital qubit channel is a mixture of unitary channels, as the tetrahedron shows, but for $N\ge3$ there are unital channels that are not \citep{landau1993birkhoff}; the mixtures of unitaries are those produced by the random Schr\"odinger equation of \Chap~\ref{C.Open} \citep{lawrence2026geometric}.}
\onp{\begin{itemize}
    \item Unital qubit channels $=$ rotations $\circ$ Pauli channels $\circ$ rotations; coordinates $\vec\lambda$ \citep{king2001minimal}.
    \item Positive: the cube. CP: the tetrahedron of $\Id,\sigma_x,\sigma_y,\sigma_z$ \citep{fujiwara1999one,ruskai2002analysis}. Transposition: a corner outside.
    \item Via Choi: the Bell-diagonal tetrahedron of \Chap~\ref{C.GeometryStates}; the separable octahedron $=$ entanglement-breaking channels \citep{horodecki2003entanglement,ruskai2003qubit}.
\end{itemize}}
\end{frame}

\section{Distances between channels}\label{sec:ch08:distances}

\begin{frame}\ft{The diamond norm}
\ona{How well can two channels $\Phi$ and $\Psi$ be told apart if we may use each once? We may prepare any input, and Helstrom's theorem (Theorem~\ref{thm:ch07:helstrom}) tells us how well the two outputs can be distinguished; but the input may also be entangled with an ancilla that the channel does not touch, and this can help.}
\begin{defn}[Diamond norm]\label{def:ch08:diamond}
For a linear map $\Delta$ on the operators of $\C^N$,
\[
    \dnorm\Delta\coloneqq\max_{\rho\in\Dens[\C^N\otimes\C^N]}\tracenorm{(\Delta\otimes\mathrm{id}_N)(\rho)} .
\]
The \emph{diamond distance} of two channels is $\tfrac12\dnorm{\Phi-\Psi}$.
\end{defn}
\ona{By Helstrom's theorem, the greatest probability of naming correctly, in one use, which of two equally likely channels was applied is $\tfrac12\big(1+\tfrac12\dnorm{\Phi-\Psi}\big)$. The norm was introduced by \citet{kitaev1997quantum}, who showed that an ancilla of the dimension $N$ of the system is always enough, and that the norm is stable: adding further idle systems does not increase it. It can be computed by a semidefinite programme \citep{watrous2009semidefinite}. Without the ancilla one obtains a smaller quantity, and the difference can be large. For the qubit maps $\Phi_\pm(\rho)=\tfrac{1}{2\pm1}\big(\Tr(\rho)\Id\pm\rho^\top\big)$ of \citet{werner2002counterexample}, which are channels, no single-qubit input makes their outputs further apart than $\tfrac23$ in trace distance; but sent through one half of the Bell state $\ket{\Phi^+}$ they produce the singlet and the normalised projector onto its orthogonal complement, which are perfectly distinguishable, so that $\tfrac12\dnorm{\Phi_+-\Phi_-}=1$ (Figure~\ref{fig:ch08:diamond}(a)).}
\onp{\begin{itemize}
    \item One use, best guess: success $\tfrac12(1+\tfrac12\dnorm{\Phi-\Psi})$; ancilla of dimension $N$ suffices; stable \citep{kitaev1997quantum}; an SDP \citep{watrous2009semidefinite}.
    \item The ancilla matters: for the Werner--Holevo pair \citep{werner2002counterexample}, $\tfrac23$ without it, $1$ with it.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Average case and worst case}
\ona{In control and in experiments one more often uses fidelities, which are averages. The \emph{process fidelity} of a channel $\Phi$ with a unitary target $U$ is the fidelity of the normalised Choi states, $\Fid_{\mathrm{pro}}=\Fid\big(\Choi\Phi/N,\Choi{U\cdot U^\dagger}/N\big)=\tfrac1{N^2}\sum_k\abs{\Tr(U^\dagger K_k)}^2$, and the \emph{average gate fidelity} is the fidelity of the output with the target output, averaged over pure inputs drawn uniformly,}
\onp{Process fidelity $\Fid_{\mathrm{pro}}=\tfrac1{N^2}\sum_k\abs{\Tr(U^\dagger K_k)}^2$; average gate fidelity \citep{horodecki1999general,nielsen2002simple}:}
\begin{equation}\label{eq:ch08:avgfid}
    \bar\Fid(\Phi,U)=\int\braket{\psi|U^\dagger\Phi(\ket\psi\bra\psi)U|\psi}\,\mathrm d\psi=\frac{N\Fid_{\mathrm{pro}}+1}{N+1}
\end{equation}
\ona{\citep{horodecki1999general,nielsen2002simple}. For a unitary channel $V\cdot V^\dagger$ the process fidelity is $\abs{\Tr(U^\dagger V)}^2/N^2$, the gate fidelity used as an objective in \Chaps~\ref{C.Motivation}, \ref{C.GeometryControl} and~\ref{C.AlgControl}. Since the Choi state of a unitary target is pure, the Fuchs--van de Graaf inequalities of \Chap~\ref{C.GeometryStates} and the comparison of the diamond norm with the trace norm of Choi states give}
\onp{Bounds, from Fuchs--van de Graaf (\Chap~\ref{C.GeometryStates}):}
\begin{equation}\label{eq:ch08:dnbounds}
    1-\Fid_{\mathrm{pro}}\ \le\ \tfrac12\dnorm{\Phi-U\cdot U^\dagger}\ \le\ N\sqrt{1-\Fid_{\mathrm{pro}}} .
\end{equation}
\ona{The gap between the two sides is not an artefact of the proof. A stochastic error, such as $\sigma_z$ applied with the probability $p$, has a diamond distance from the identity proportional to its infidelity, but a coherent error, such as the over-rotation $\ee^{-\imath\theta\sigma_z/2}$, has a diamond distance proportional to the square root of its infidelity. At an average infidelity $r=1-\bar\Fid=10^{-4}$, typical of a good gate today, the stochastic error has the diamond distance $1.5\cdot10^{-4}$ and the coherent one $1.2\cdot10^{-2}$ (Figure~\ref{fig:ch08:diamond}(b)). The thresholds of fault-tolerant quantum computation are stated in the diamond norm, while experiments report average fidelities, and the conversion between the two depends on the nature of the error \citep{sanders2016bounding,kueng2016comparing}.}
\onp{\begin{itemize}
    \item Stochastic errors: diamond $\propto$ infidelity. Coherent errors: diamond $\propto\sqrt{\text{infidelity}}$.
    \item At $r=10^{-4}$: $1.5\cdot10^{-4}$ against $1.2\cdot10^{-2}$; thresholds are stated in the diamond norm \citep{sanders2016bounding,kueng2016comparing}.
\end{itemize}}
\end{frame}

\section{Divisibility and Markovianity}\label{sec:ch08:markov}

\begin{frame}\ft{Why the classical definition does not transfer}
\ona{When computing the dynamics of an open quantum system, assumptions are often made in order to simplify the description and facilitate the efficient modelling of the system. One of the best known is the Markovian property \citep{Bhattacharya2016}. Intuitively, a system's evolution is called Markovian in the classical sense if the evolution of the state depends only on the state at the current time: the system has no history and is memoryless. Mathematically, a stochastic process is Markovian if}
\begin{equation}\label{eq:ch08:classical-markov}
    P(x_n,t_n\,|\,x_{n-1},t_{n-1};\dots;x_0,t_0)=P(x_n,t_n\,|\,x_{n-1},t_{n-1}),\qquad t_n>t_{n-1}>\dots>t_0 .
\end{equation}
\ona{In the context of quantum systems this definition is problematic. The left-hand side conditions on a series of measurements, which are device-dependent perturbations introduced into the system when extracting data; the definition is thus inherently dependent on the measurement, which partially collapses the system and changes it significantly. One must therefore look for another description of a quantum Markovian process. There is no consensus on such a definition: many criteria and related concepts have been proposed, none of which captures the memorylessness of open quantum systems fully. They fall into two categories \citep{LI20181}: those that consider only the dynamics of the reduced system \citep{rivas_14,Breuer2016,Breuer_09}, and those that consider its interaction with the environment. The choice is context dependent; we discuss divisibility, which belongs to the first category and leads to the common Lindbladian definition of Markovianity.}
\onp{\begin{itemize}
    \item Classical: the future depends on the past only through the present.
    \item Quantum: conditioning on the past means \emph{measuring} it, which disturbs the system.
    \item No consensus \citep{LI20181}; we follow the reduced dynamics: divisibility \citep{rivas_14,Breuer2016}.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Divisibility}
\ona{A classical process is called divisible if its transition matrices compose, $T(t_2|t_0)=T(t_2|t_1)T(t_1|t_0)$, with every factor stochastic. All Markovian processes are divisible, but not conversely \citep{Markov_div_13,Vacchini_2011}; divisibility is the ability to break the evolution into intermediate steps, without implying that the process is memoryless in the sense of \eqref{eq:ch08:classical-markov}. The quantum analogue replaces the transition matrices by the dynamical maps.}
\begin{defn}[P- and CP-divisibility]\label{def:ch08:divisible}
A family of channels $\{\Phi_t\}_{t\ge0}$ is \emph{P-divisible} if for all $t\ge s\ge0$ there is a positive trace-preserving map $\mathcal E_{t,s}$ with $\Phi_t=\mathcal E_{t,s}\Phi_s$, and \emph{CP-divisible} if every $\mathcal E_{t,s}$ can be taken completely positive \citep{rivas_14}.
\end{defn}
\begin{propn}[Nested images]\label{prop:ch08:nested}
If $\{\Phi_t\}$ is P-divisible, then $\Phi_t(\Dens)\subseteq\Phi_s(\Dens)$ and $D(\Phi_t\rho,\Phi_t\sigma)\le D(\Phi_s\rho,\Phi_s\sigma)$ for all $t\ge s$ and all states $\rho,\sigma$.
\end{propn}
\ona{\begin{pf}
$\Phi_t(\Dens)=\mathcal E_{t,s}\big(\Phi_s(\Dens)\big)\subseteq\Phi_s(\Dens)$, since $\mathcal E_{t,s}$ maps states to states; and the trace distance cannot increase under the positive trace-preserving map $\mathcal E_{t,s}$ (Theorem~\ref{thm:ch07:contract}).
\end{pf}}
\ona{The second statement is the criterion of \citet{Breuer_09}: an increase of the trace distance between two evolving states, a flow of information back from the environment, rules out divisibility. The first gives a picture. For a qubit the image $\Phi_t(\Dens)$ is an ellipsoid in the Bloch ball, and a divisible family of channels has ellipsoids that shrink into one another as time goes on (Figure~\ref{fig:ch08:nested}(a)). For the qubit with the hidden defect of \Chap~\ref{C.StateSpace}, the reduced channel is amplitude damping with the time-dependent amplitude $c(t)$ of \Chap~\ref{C.GeometryStates}, and its image is the ellipse with the semi-axes $\abs{c(t)}$ and $c(t)^2$. At $t=1.81$, where $c$ vanishes, the image collapses to the single point $\ket0$, and afterwards it grows again (Figure~\ref{fig:ch08:nested}(b)). No map whatever, positive or not, can turn a point into an ellipse, so this family is not divisible at all: the excitation that returns from the defect carries information that the qubit alone had lost.}
\onp{\begin{itemize}
    \item P-divisible $\Rightarrow$ images nested and trace distance non-increasing \citep{Breuer_09}.
    \item Hidden defect (\Chap~\ref{C.StateSpace}): the image collapses to $\ket0$ at $t=1.81$, then grows: not divisible at all.
\end{itemize}}
\end{frame}

\begin{frame}\ft{CP-divisibility and time-local master equations}
\begin{propn}[\citealp{wolf06,Rivas2011OpenQS,Kossakowski10}]\label{prop:ch08:cpdiv}
Let $\{\Phi_t\}_{t\ge0}$ be a family of channels, differentiable in $t$ and invertible for every $t$. It is CP-divisible if and only if it satisfies a master equation of the form
\begin{equation}\label{eq:ch08:timelocal}
    \dot\rho(t)=-\imath\comm{H(t)}{\rho(t)}+\sum_k\gamma_k(t)\Big[V_k(t)\rho(t)V_k(t)^\dagger-\tfrac12\acomm{V_k(t)^\dagger V_k(t)}{\rho(t)}\Big]
\end{equation}
with $H(t)$ Hermitian and all rates $\gamma_k(t)\ge0$.
\end{propn}
\ona{The generator is $\Lind(t)=\dot\Phi_t\Phi_t^{-1}$, which is where invertibility is needed; the canonical form of $\Lind(t)$, with the rates $\gamma_k(t)$ as the eigenvalues of a Kossakowski matrix, is that of Theorem~\ref{thm:ch03:gksl} \citep{Hall2014}. The rates of the defect are $\gamma(t)=-2\,\tfrac{\mathrm d}{\mathrm dt}\ln\abs{c(t)}$, negative while $\abs c$ grows and infinite where $c$ vanishes, so that the generator does not exist at those instants (Figure~\ref{fig:ch08:nested}(c)); this is the non-Markovianity foreshadowed in \Chap~\ref{C.Open}. Positivity of the intermediate maps is weaker than complete positivity, and P-divisible families with a rate that is negative for all $t>0$ exist: for the Pauli dynamics with the rates $\gamma_1=\gamma_2=1$ and $\gamma_3(t)=-\tanh t$ of \citet{Hall2014}, every $\Phi_t$ is a channel and the family is P-divisible, but it is not CP-divisible at any time. CP-divisibility is one of several possible definitions of quantum Markovianity, and it does not imply Markovianity in the operational, multi-time sense \citep{Milz2019}. Geometrically, the CP-divisible families are the curves in $\Chan$ whose logarithmic derivative $\dot\Phi_t\Phi_t^{-1}$ lies at every instant in the convex cone of GKSL generators.}
\onp{\begin{itemize}
    \item $\Lind(t)=\dot\Phi_t\Phi_t^{-1}$ in canonical form \citep{Hall2014}; negative rate $\Leftrightarrow$ not CP-divisible.
    \item Defect: $\gamma(t)=-2\tfrac{\mathrm d}{\mathrm dt}\ln\abs{c(t)}$, negative while the image grows.
    \item P- but not CP-divisible: $\gamma_1=\gamma_2=1$, $\gamma_3(t)=-\tanh t$ \citep{Hall2014}.
    \item CP-divisible $\not\Rightarrow$ Markovian in the multi-time sense \citep{Milz2019}.
\end{itemize}}
\end{frame}

\begin{frame}\ft{Dynamical semigroups}
\ona{A stronger requirement than divisibility is invariance under translations in time. A family is \emph{time homogeneous} if $\mathcal E_{t,s}$ depends only on $t-s$, and then the composition law reduces to}
\begin{equation}\label{eq:ch08:semigroup}
    \Phi_{\tau_1}\Phi_{\tau_2}=\Phi_{\tau_1+\tau_2}\qquad\tau_1,\tau_2\ge0 .
\end{equation}
\ona{A \emph{quantum dynamical semigroup} is a family of channels satisfying \eqref{eq:ch08:semigroup} with $\Phi_0=\mathrm{id}$, and with $\Phi_\tau(X)\to X$ as $\tau\to0^+$ for every $X$. The continuity guarantees a generator: $\Phi_\tau=\ee^{\tau\Lind}$ with $\Lind$ of the GKSL form with a constant Hamiltonian and a constant Kossakowski matrix $\gamma\succeq0$ \citep{gorini1976completely,Lindblad1976}. In the coordinates of the coherence vector this is the autonomous linear system $\dot{\vec x}=\mathbf A\vec x+\vec\beta$ of \Chap~\ref{C.Open}, and the identification of $\Lind$ from data is the subject of \Chaps~\ref{C.StateSpace}--\ref{C.SampleComplexity}. The qubit of \Chap~\ref{C.Frequency} is a semigroup; the dephasing by Ornstein--Uhlenbeck noise of \Chap~\ref{C.Open}, whose rate $b_{\rm rms}^2\tau_c\big(1-\ee^{-t/\tau_c}\big)$ is non-negative but grows with time, is CP-divisible but not a semigroup, and becomes one only in the white-noise limit. Figure~\ref{fig:ch08:families} places the examples of this section in the hierarchy of families.}
\onp{\begin{itemize}
    \item Time homogeneous $+$ divisible: $\Phi_\tau=\ee^{\tau\Lind}$, constant GKSL generator \citep{gorini1976completely,Lindblad1976}: $\dot{\vec x}=\mathbf A\vec x+\vec\beta$.
    \item Ornstein--Uhlenbeck dephasing (\Chap~\ref{C.Open}): rate $\ge0$ but time dependent; CP-divisible, not a semigroup.
\end{itemize}}
\end{frame}

\section{Groups and semigroups of channels}\label{sec:ch08:lie}

\begin{frame}\ft{Unitary channels: a Lie group}
\ona{The unitary channels $\rho\mapsto U\rho U^\dagger$ form a copy of the projective unitary group $\mathrm{PU}(N)$, the unitaries up to a phase, inside $\Chan$, and they lie among its extreme points (Theorem~\ref{thm:ch08:extreme}). As a compact Lie group it carries the bi-invariant Riemannian metric $\inner XY=\Tr(X^\dagger Y)$ on $\liesu[N]$, whose geodesics through $\Id$ are the one-parameter groups $t\mapsto\ee^{-\imath Ht}$; the distance from $\Id$ to $U$ is the length of the vector of the eigenphases of $U$, taken in $(-\pi,\pi]$ and shifted by a common phase to make it shortest. In \Chap~\ref{C.GeometryControl} the controls restrict the velocities $-\imath H$ to the drift and the span of the control Hamiltonians, which replaces this Riemannian geometry by a sub-Riemannian one and makes the length of a trajectory, and hence the least time, a measure of the complexity of $U$ (\Chap~\ref{C.Complexity}).}
\onp{\begin{itemize}\setlength\itemsep{0.5em}
    \item Unitary channels $\cong\mathrm{PU}(N)$: a compact Lie group, among the extreme points of $\Chan$.
    \item Bi-invariant metric $\Tr(X^\dagger Y)$; geodesics $\ee^{-\imath Ht}$; distance from the eigenphases.
    \item Restricted velocities: sub-Riemannian geometry (\Chap~\ref{C.GeometryControl}); length and complexity (\Chap~\ref{C.Complexity}).
\end{itemize}}
\end{frame}

\begin{frame}\ft{Markovian channels: a Lie semigroup}
\ona{The channels of the form $\ee^{\Lind}$, with $\Lind$ a GKSL generator, and more generally those reached by CP-divisible families, are closed under composition and contain the identity, and they are generated by the convex cone of GKSL generators; they form a \emph{Lie semigroup}. They are not closed under mixtures. For Pauli channels the semigroup is the curved region $\lambda_i\ge\lambda_j\lambda_k$ of the positive octant, and the midpoint of the Markovian channels $(0.9,0.3,0.3)$ and $(0.3,0.9,0.3)$ lies outside it, since its rate $\gamma_3$ would have to be negative (Figure~\ref{fig:ch08:pauli}(b)) \citep{wolf2008assessing,davalos2019divisibility}. Nor do they include every channel. The determinant of $\ee^{\Lind}$ is positive, so that a qubit channel whose matrix $M$ has a negative determinant, such as $\Bloch\mapsto-\tfrac13\Bloch$, the best physical approximation to the impossible inversion of the Bloch ball \citep{buzek1999optimal}, is not of the form $\ee^{\Lind}$; some channels cannot even be divided into two channels that are not unitary \citep{wolf06}. Deciding whether a given channel is Markovian in this sense is NP-hard \citep{cubitt2012complexity}.}
\ona{Two consequences matter in this course. Controlled GKSL dynamics, $\dot{\vec x}=(\mathbf A+\sum_ku_k\mathbf N_k)\vec x+\vec\beta$, is a bilinear system on a semigroup rather than on a group, so that controllability must be replaced by accessibility and the reachable sets are not symmetric in time (\Chap~\ref{C.StateSpace}). And fitting a GKSL model to data is a constrained problem whose feasible set is not convex, which is one motivation for the polynomial optimisation of \Chap~\ref{C.AlgTomography}.}
\onp{\begin{itemize}\setlength\itemsep{0.4em}
    \item $\{\ee^{\Lind}\}$: a Lie semigroup generated by the convex GKSL cone; not convex, as the Pauli semigroups showed \citep{wolf2008assessing,davalos2019divisibility}.
    \item $\det\ee^{\Lind}>0$: $\Bloch\mapsto-\tfrac13\Bloch$ is not Markovian; deciding Markovianity is NP-hard \citep{cubitt2012complexity}.
    \item For us: accessibility, not controllability (\Chap~\ref{C.StateSpace}); GKSL fitting is non-convex (\Chap~\ref{C.AlgTomography}).
\end{itemize}}
\end{frame}

\begin{frame}\ft{Summary}\mop{\small}
\begin{itemize}\setlength\itemsep{0.4em}
    \item The Choi matrix turns a channel into a positive semidefinite matrix; its eigenvectors give the Kraus operators, and the Stinespring dilation makes every channel a unitary evolution with an environment that is discarded.
    \item Channels form a spectrahedron of dimension $N^4-N^2$. The unital qubit channels form a tetrahedron, the Bell-diagonal tetrahedron in disguise, with the entanglement-breaking channels as its octahedron.
    \item The diamond norm measures the worst case and needs an ancilla; fidelities measure the average case. For coherent errors the two differ as an infidelity and its square root.
    \item Divisible families have nested images; CP-divisibility means a time-local master equation with non-negative rates, and semigroups are its autonomous case.
    \item Unitary channels form a Lie group; Markovian channels a Lie semigroup, which is not convex.
    \item Next: optimal control on these groups and semigroups.
\end{itemize}
\end{frame}
